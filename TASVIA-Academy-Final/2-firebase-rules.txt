rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null && request.auth.token.email != null;
    }
    function staffDoc() {
      return exists(/databases/$(database)/documents/Staff/$(request.auth.token.email.lower()));
    }
    function isStaff() {
      return signedIn()
        && staffDoc()
        && get(/databases/$(database)/documents/Staff/$(request.auth.token.email.lower())).data.role in ["staff", "admin"];
    }
    function isAdmin() {
      return isStaff()
        && get(/databases/$(database)/documents/Staff/$(request.auth.token.email.lower())).data.role == "admin";
    }

    match /Staff/{email} {
      allow read: if signedIn()
        && request.auth.token.email.lower() == email.lower()
        && isStaff();
      allow write: if false;
    }

    match /parentView/{code} {
      allow get: if true;
      allow list, write: if isStaff();
    }

    match /students/{docId} {
      allow read: if isStaff();
      allow write: if isAdmin();
    }

    match /fees/{docId} {
      allow read, write: if isAdmin();
    }

    match /teachers/{docId} {
      allow read, write: if isAdmin();
    }

    match /salaries/{docId} {
      allow read, write: if isAdmin();
    }

    match /expenses/{docId} {
      allow read, write: if isAdmin();
    }

    match /info/{docId} {
      allow read: if isStaff();
      allow write: if isAdmin();
    }

    match /attendance/{docId} {
      allow read, write: if isStaff();
    }

    match /staffattendance/{docId} {
      allow read, write: if isStaff();
    }

    match /tests/{docId} {
      allow read, write: if isStaff();
    }

    match /diary/{docId} {
      allow read, write: if isStaff();
    }

    match /{collection}/{docId} {
      allow read, write: if false;
    }
  }
}
